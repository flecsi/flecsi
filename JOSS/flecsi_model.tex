\def\KernelSpacing{0.65}
\def\GroupSpacing{0.45}
\def\LeftOffset{0.6}

% Kernels for pt1-pt4 (two each: k1 to k8)
\foreach \i [evaluate=\i as \groupX using (\i-1)*\GroupSpacing] in {1,...,4}{
  \foreach \j [evaluate=\j as \localshift using (\j-1)*\KernelSpacing] in {1,2}{
    \pgfmathtruncatemacro{\knumber}{2*(\i-1)+\j}
    \pgfmathsetmacro{\xshift}{\groupX+\localshift - (\KernelSpacing/2) - \LeftOffset}
    \node[kernel] (k\knumber) at ([xshift=\xshift cm,yshift=-1cm]pt\i.south) {K};
    \draw[arrow] (pt\i.south) -- (k\knumber.north);
  }
}
